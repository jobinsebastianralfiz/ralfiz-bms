// --- App.Formulas ---
ThemeBlue = ColorValue("#1F4E9A");

MeUser = LookUp(Users, 'Primary Email' = User().Email);

MyOpenTickets = Filter(
    Tickets,
    'Created By'.User = MeUser.User && Status <> 'Status (Tickets)'.Closed
);

OpenTicketCount = CountRows(Filter(Tickets, Status = 'Status (Tickets)'.Open));

DueLabel(d: Date): Text =
    If(d < Today(), "Overdue", "Due " & Text(d, "dd mmm"));

PriorityColour(p: Text): Color =
    Switch(p,
        "High", ColorValue("#B42318"),
        "Medium", ColorValue("#B54708"),
        ColorValue("#475467"));

// --- App.OnStart (AFTER) ---
// Constants and derived values moved to App.Formulas. What must stay runs in parallel.
Concurrent(
    ClearCollect(colCategories, Categories),
    Set(varMe, User())
)

// --- galTickets.Items ---
SortByColumns(MyOpenTickets, "chd_duedate", SortOrder.Ascending)

// --- galTickets / lblDue.Text ---
DueLabel(ThisItem.'Due Date')

// --- galTickets / lblPriority.Text ---
Text(ThisItem.Priority)
// --- galTickets / lblPriority.Color ---
PriorityColour(Text(ThisItem.Priority))

// --- lblOpen.Text ---
OpenTicketCount & " open tickets across all campuses"
