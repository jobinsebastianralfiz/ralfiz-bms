// B.15 row-level security filters (DAX). Modeling > Manage roles.
// In the role editor switch to the DAX editor (or use the table filter boxes) and paste one filter per role.

// ===== Static roles: table Tickets =====
// Role North
[Campus] = "North"
// Role South
[Campus] = "South"
// Role City
[Campus] = "City"

// ===== Dynamic role 1: Own Tickets, table Staff =====
// Needs Staff (from staff.csv) related Staff[AssignedTo] 1 -> * Tickets[AssignedTo], single direction.
[Email] = USERPRINCIPALNAME ()

// ===== Dynamic role 2: Campus Lead, table Tickets =====
// Needs StaffCampus (from staff-campus.csv) loaded with NO relationship to other tables.
// One person can map to several campuses (Suresh maps to all three).
[Campus]
    IN CALCULATETABLE (
        VALUES ( StaffCampus[Campus] ),
        StaffCampus[Email] = USERPRINCIPALNAME ()
    )
